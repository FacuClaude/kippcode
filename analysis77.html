from pathlib import Path import re import unicodedata from main import read_document PROJECT_ROOT = Path(__file__).resolve().parent.parent DOCUMENTS_FOLDER = PROJECT_ROOT / "documents" SUPPORTED_EXTENSIONS = {".pdf", ".docx", ".xlsx", ".pptx"} def normalize(text): text = text.lower() text = unicodedata.normalize("NFD", text) text = "".join( char for char in text if unicodedata.category(char) != "Mn" ) return re.findall(r"\b\w+\b", text) def create_chunks(text, chunk_size=500): paragraphs = [ paragraph.strip() for paragraph in text.split("\n") if paragraph.strip() ] chunks = [] current_chunk = "" for paragraph in paragraphs: if len(current_chunk) + len(paragraph) <= chunk_size: current_chunk += paragraph + "\n" else: if current_chunk: chunks.append(current_chunk.strip()) current_chunk = paragraph + "\n" if current_chunk: chunks.append(current_chunk.strip()) return chunks def calculate_score(question_words, chunk, question): chunk_words = normalize(chunk) chunk_normalized = " ".join(chunk_words) stop_words = { "que", "como", "cual", "cuales", "cuanto", "cuantos", "cuanta", "cuantas", "de", "del", "la", "las", "el", "los", "un", "una", "unos", "unas", "y", "o", "en", "por", "para", "con", "puede", "pueden", "tener", "tiene", "empleado", "empleados" } important_words = { word for word in question_words if word not in stop_words and len(word) > 2 } score = 0 # Coincidencias de palabras importantes for word in important_words: if word in chunk_words: score += 3 # Coincidencias de frases de dos palabras question_list = normalize(question) for i in range(len(question_list) - 1): phrase = ( question_list[i] + " " + question_list[i + 1] ) if phrase in chunk_normalized: score += 10 return score def search_documents(question): question_words = set(normalize(question)) results = [] for file in DOCUMENTS_FOLDER.iterdir(): if ( not file.is_file() or file.suffix.lower() not in SUPPORTED_EXTENSIONS ): continue try: text = read_document(file) if not text: continue chunks = create_chunks(text) for chunk_number, chunk in enumerate( chunks, start=1 ): score = calculate_score( question_words, chunk, question ) if score > 0: results.append( { "file": file.name, "chunk": chunk_number, "score": score, "text": chunk } ) except Exception as error: print( f"Error leyendo {file.name}: {error}" ) results.sort( key=lambda result: result["score"], reverse=True ) return results def main(): print() print("=== CORPORATE AI — SEARCH V3 ===") print() question = input("¿Qué querés saber? ") results = search_documents(question) print() if not results: print( "No encontré información relacionada " "con la pregunta." ) return print("Resultados más relevantes:") print() for position, result in enumerate( results[:5], start=1 ): print("=" * 60) print(f"RESULTADO {position}") print(f"Fuente: {result['file']}") print(f"Fragmento: {result['chunk']}") print(f"Relevancia: {result['score']}") print("-" * 60) print(result["text"]) print() if __name__ == "__main__": main()
