from pathlib import Path from pypdf import PdfReader # Ubicación de la carpeta de documentos PROJECT_ROOT = Path(__file__).resolve().parent.parent DOCUMENTS_FOLDER = PROJECT_ROOT / "documents" def read_pdf(pdf_path): """Extrae el texto de un archivo PDF.""" reader = PdfReader(str(pdf_path)) text = [] for page in reader.pages: text.append(page.extract_text() or "") return "\n".join(text) def main(): print("=== CORPORATE AI — Document Reader ===") print(f"Carpeta de documentos: {DOCUMENTS_FOLDER}") print() pdf_files = list(DOCUMENTS_FOLDER.glob("*.pdf")) if not pdf_files: print("No se encontraron archivos PDF.") print("Colocá un PDF de prueba dentro de la carpeta documents.") return for pdf_path in pdf_files: print(f"Archivo: {pdf_path.name}") try: text = read_pdf(pdf_path) print(f"Caracteres extraídos: {len(text)}") print("-" * 50) print(text[:1000]) print("-" * 50) except Exception as error: print(f"Error al leer el archivo: {error}") if __name__ == "__main__": main()
